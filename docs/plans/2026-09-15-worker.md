# Transcriber Worker Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Python worker that turns two growing PCM tracks (mic = Владислав, sys = everyone else) into a live-updated verbatim transcript in the Obsidian vault, with speaker labels, echo removal, crash-safe resume, and a finalization pass; plus a `from-wav` CLI that runs the same pipeline on recorded files.

**Architecture:** Pure functions and small classes with injected models (VAD probability callable, ASR callable, embedding callable) so every module is unit-tested with fakes and only one integration path touches real models. State lives in two JSON files per session: `meta.json` (owned by the Swift app: who/when/what) and `state.json` (owned by the worker: offsets, records, embeddings). A `STOP` marker file in the session dir requests finalization. The vault Raw file is rewritten atomically on every update; the MTG stub is created once.

**Tech Stack:** Python 3.12 via uv (project already at `~/Claude/transcriber/worker`, deps installed: mlx-whisper 0.4.3, silero-vad 6.2, pyannote.audio 4.0.7, torch 2.14 + MPS, numpy, pyyaml, python-dotenv, pytest). Add `scipy` (resampling + hierarchical clustering).

Spec: `docs/specs/2026-09-15-transcriber-design.md`. Deviation from spec locked in here: `session.json` is split into `meta.json` (Swift-owned) + `state.json` (worker-owned) + `STOP` marker, so two processes never write the same file. Task 0 records this in the spec.

All commands run from `~/Claude/transcriber/worker` unless stated. Commit from `~/Claude/transcriber` (repo root). Commit messages end with `Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>`.

**Post-plan additions:** по данным первой живой встречи — `speaker_similarity` 0.3, подклейка кластеров-одиночек, второе правило эхо-фильтра (`echo_strong_similarity`), модуль `quality.py` с фильтрами галлюцинаций whisper, отсев слабых сегментов в `Transcriber`, создание Raw и MTG-заглушки в демоне до загрузки моделей и команда `transcriber refinalize <session_id>` — постобработка сохранённого `state.json` без аудио. **Post-plan additions (голосовые профили):** метки говорящих переносятся между встречами. Модуль `voices.py` — библиотека профилей в `voices/` (по JSON на человека: центроид как среднее единичных векторов, до 50 свежих сэмплов, источники), новые настройки `voices_dir`, `voice_match_similarity: 0.45`, `voice_min_samples: 3`. def tone(seconds: float, freq: float = 440.0, amp: float = 0.3, sr: int = SR) -> np.ndarray:
    t = np.arange(int(seconds * sr)) / sr
    return (amp * np.sin(2 * np.pi * freq * t)).astype(np.float32)


def silence(seconds: float, sr: int = SR) -> np.ndarray:

class FakeAsr:
    """Returns a label derived from the dominant frequency so tests can tell utterances apart."""

    def __init__(self, fail_on: set[int] | None = None):
        self.fail_on = fail_on or set()
        self.calls: list[tuple[int, str]] = []

    def __call__(self, audio: np.ndarray, prompt: str = "") -> str:
        f = dominant_freq(audio)
        self.calls.append((f, prompt))
        if f in self.fail_on:

        return f"tone{f}"


def dominant_freq(audio: np.ndarray, sr: int = SR) -> int:

```python
from __future__ import annotations

from pathlib import Path

import numpy as np

BYTES_PER_SAMPLE = 2  # int16 mono


class PcmReader:
    """Reads int16 mono PCM from a file that may still be growing, starting at a sample offset."""

    def __init__(self, path: Path, offset_samples: int = 0):
        self.path = Path(path)
        self.offset = int(offset_samples)

    def available(self) -> int:
        if not self.path.exists():
            return 0
        return max(0, self.path.stat().st_size // BYTES_PER_SAMPLE - self.offset)

    def read_all(self) -> np.ndarray:
        n = self.available()
        if n <= 0:
            return np.zeros(0, dtype=np.float32)
        with open(self.path, "rb") as f:
            f.seek(self.offset * BYTES_PER_SAMPLE)
            raw = f.read(n * BYTES_PER_SAMPLE)
        data = np.frombuffer(raw, dtype=np.int16)
        self.offset += len(data)
        return data.astype(np.float32) / 32768.0
```

```python
from __future__ import annotations

from dataclasses import dataclass
from typing import Callable

import numpy as np

CHUNK = 512  # silero expects 512 samples at 16 kHz


@dataclass
class Utterance:
    start: int  # absolute sample index
    end: int
    audio: np.ndarray

    @property
    def samples(self) -> int:
        return self.end - self.start


class Segmenter:
    def __init__(self, speech_prob: Callable[[np.ndarray], float], sample_rate: int = 16000,
                 threshold: float = 0.5, min_silence_ms: int = 700, speech_pad_ms: int = 100,
                 max_utterance_s: float = 30.0, start_sample: int = 0):
        self.prob = speech_prob
        self.sr = sample_rate
        self.threshold = threshold
        self.min_silence = int(sample_rate * min_silence_ms / 1000)
        self.pad = int(sample_rate * speech_pad_ms / 1000)
        self.max_len = int(sample_rate * max_utterance_s)
        self.pos = int(start_sample)          # absolute index of the next chunk to consume
        self.remainder = np.zeros(0, dtype=np.float32)
        self.open_start: int | None = None    # absolute start of the open utterance (pad applied)
        self.silence_run = 0                  # samples of silence inside the open utterance
        self.buf: list[np.ndarray] = []       # chunks since buf_start
        self.buf_start = self.pos
        self.tail = np.zeros(0, dtype=np.float32)  # last chunk before speech, for the pad

    def boundary(self) -> int:
        return self.open_start if self.open_start is not None else self.pos

    def feed(self, audio: np.ndarray) -> list[Utterance]:
        data = np.concatenate([self.remainder, audio.astype(np.float32)])
        n_full = len(data) // CHUNK
        self.remainder = data[n_full * CHUNK:]
        out: list[Utterance] = []
        for i in range(n_full):
            chunk = data[i * CHUNK:(i + 1) * CHUNK]
            out.extend(self._consume(chunk))
        return out

    def _consume(self, chunk: np.ndarray) -> list[Utterance]:
        speech = self.prob(chunk) >= self.threshold
        out: list[Utterance] = []
        if self.open_start is None:
            if speech:
                pad_avail = min(self.pad, len(self.tail))
                self.open_start = self.pos - pad_avail
                self.buf_start = self.pos - len(self.tail)
                self.buf = [self.tail, chunk] if len(self.tail) else [chunk]
                self.silence_run = 0
            else:
                self.tail = np.concatenate([self.tail, chunk])[-self.pad:]
            self.pos += CHUNK
            return out
        self.buf.append(chunk)
        self.pos += CHUNK
        if speech:
            self.silence_run = 0
        else:
            self.silence_run += CHUNK
        if self.silence_run >= self.min_silence:
            end = min(self.pos - self.silence_run + self.pad, self.pos)
            out.append(self._close(end))
            self.tail = chunk[-self.pad:]
        elif self.pos - self.open_start >= self.max_len:
            out.append(self._close(self.pos))
            self.tail = np.zeros(0, dtype=np.float32)
        return out

    def _close(self, end: int) -> Utterance:
        full = np.concatenate(self.buf)
        start = self.open_start
        assert start is not None
        audio = full[start - self.buf_start:end - self.buf_start]
        self.open_start = None
        self.buf = []
        self.silence_run = 0
        return Utterance(start=start, end=end, audio=audio)
```

from __future__ import annotations

import re
from pathlib import Path

from .config import Config
from .session import Meta, Rec, Session, atomic_write_text

APP_NAMES = {"kontur.talk": "Толк", "zoom.us": "Zoom"}


def app_label(app: str) -> str:
    return APP_NAMES.get(app, "")


def default_title(meta: Meta) -> str:
    name = app_label(meta.app)
    return f"Встреча в {name}" if name else "Встреча"


def title_of(meta: Meta) -> str:
    return meta.title.strip() or default_title(meta)


def clean_title(title: str) -> str:
    return re.sub(r"\s+", " ", re.sub(r"[/\\:|#^\[\]]", " ", title)).strip()


def slug(title: str) -> str:
    return re.sub(r"-+", "-", re.sub(r"\s+", "-", clean_title(title))).strip("-")


def stamp(meta: Meta) -> tuple[str, str]:
    return meta.started_at.strftime("%Y-%m-%d"), meta.started_at.strftime("%H%M")


def raw_filename(meta: Meta) -> str:
    date, hhmm = stamp(meta)
    return f"Local-{date}-{hhmm}-{slug(title_of(meta))}.md"


def note_filename(meta: Meta) -> str:
    date, hhmm = stamp(meta)
    return f"MTG-{date} {hhmm} · {clean_title(title_of(meta))}.md"


def mmss(seconds: float) -> str:
    s = int(seconds)
    return f"{s // 60:02d}:{s % 60:02d}"


def render_raw(meta: Meta, records: list[Rec], n_speakers: int, status_text: str, marker_every_s: int = 300) -> str:
    date, _ = stamp(meta)
    title = title_of(meta)
    who = ", ".join(meta.participants) if meta.participants else "не получены"
    app = app_label(meta.app) or "ручной старт"
    lines = [
        "---", "type: raw", "source: local", f"source_date: {date}", "projects: []", "tags: [raw]", "---", "",
        f"# {title} — {meta.started_at.strftime('%Y-%m-%d %H:%M')}", "",
        f"Локальная запись ({app}). Участники: {who}. "
        f"Говорящих на канале собеседников: {n_speakers}. Статус: {status_text}. "
        "«Me» — Владислав (микрофон), «Speaker N» — собеседники (системный звук), метки по голосу.", "",
        "## Полный транскрипт", "",
    ]
    next_marker = marker_every_s
    for r in sorted(records, key=lambda r: r.start):
        while r.start >= next_marker:
            lines.append(f"— {mmss(next_marker)} —")
            lines.append("")
            next_marker += marker_every_s
        text = r.text if r.text is not None else f"[не распознано {mmss(r.start)}–{mmss(r.end)}]"
        if not text:
            continue
        lines.append(f"{r.speaker}: {text}")
        lines.append("")
    return "\n".join(lines)


def render_stub(meta: Meta, raw_stem: str) -> str:
    date, hhmm = stamp(meta)
    title = clean_title(title_of(meta))
    participants = ", ".join(["Владислав"] + [p for p in meta.participants if p != "Владислав"])
    return "\n".join([
        "---", "type: meeting", f"id: MTG-{date}-{hhmm}", f"title: {title}", f"date: {date}",
        f"participants: [{participants}]", "projects: []", f'transcript: "[[{raw_stem}]]"', "tags: [meeting, stub]", "---", "",
        f"# {title}", "",
        f"**Дата:** {date} {meta.started_at.strftime('%H:%M')} · **Участники:** {participants}", "",
        "## Мысли Владислава", "", "", "<!-- дистиллят: ожидает дейли-рутин -->", "",
    ])


def raw_path(cfg: Config, meta: Meta) -> Path:
    return cfg.vault_root / cfg.raw_subdir / raw_filename(meta)


def note_path(cfg: Config, meta: Meta) -> Path:
    return cfg.vault_root / cfg.meetings_subdir / note_filename(meta)

    st = session.state
    n_speakers = len(set(st.speaker_of_emb)) if st.speaker_of_emb else 0
    p = raw_path(cfg, session.meta)
    p.parent.mkdir(parents=True, exist_ok=True)

    st.raw_path = str(p)

        atomic_write_text(p, render_stub(session.meta, raw_path(cfg, session.meta).stem))
    session.state.note_path = str(p)
    return p

from __future__ import annotations

import re
from difflib import SequenceMatcher

from .session import Rec


def normalize(text: str) -> str:
    return " ".join(re.sub(r"[^\w\s]", "", text.lower(), flags=re.UNICODE).split())


def similarity(a: str, b: str) -> float:
    return SequenceMatcher(None, normalize(a), normalize(b)).ratio()


def overlap_fraction(m: Rec, others: list[Rec]) -> float:
    dur = m.end - m.start
    if dur <= 0:
        return 0.0
    total = sum(max(0.0, min(m.end, o.end) - max(m.start, o.start)) for o in others)
    return min(1.0, total / dur)


def is_echo(m: Rec, others: list[Rec], overlap: float, sim: float) -> bool:
    """A mic utterance is an echo of the speakers if it overlaps sys speech and says the same thing."""
    if not m.text:
        return False
    touching = [o for o in others if o.text and min(m.end, o.end) - max(m.start, o.start) > 0]
    if overlap_fraction(m, touching) < overlap:
        return False
    return max((similarity(m.text, o.text) for o in touching), default=0.0) >= sim

    def from_history(cls, similarity: float, embeddings: list[list[float]], labels: list[int]) -> "OnlineClusterer":
        c = cls(similarity)
        for e, l in zip(embeddings, labels):
            while len(c.sums) <= l:
                c.sums.append(np.zeros(len(e), dtype=np.float32))
                c.counts.append(0)
            c.sums[l] = c.sums[l] + _unit(np.asarray(e))
            c.counts[l] += 1
        return c

    @property
    def n_speakers(self) -> int:
        return len(self.sums)

    def assign(self, emb: np.ndarray) -> int:
        e = _unit(emb)
        if self.sums:
            sims = [float(np.dot(e, _unit(s))) for s in self.sums]
            best = int(np.argmax(sims))
            if sims[best] >= self.similarity:
                self.sums[best] = self.sums[best] + e
                self.counts[best] += 1
                return best
        self.sums.append(e.copy())
        self.counts.append(1)
        return len(self.sums) - 1


def recluster(embeddings: np.ndarray, similarity: float) -> list[int]:
    """Average-linkage agglomerative clustering on cosine distance; labels ordered by first appearance."""
    n = len(embeddings)
    if n == 0:
        return []
    if n == 1:
        return [0]
    x = np.stack([_unit(e) for e in embeddings])
    z = linkage(x, method="average", metric="cosine")
    raw = fcluster(z, t=1.0 - similarity, criterion="distance")
    order: dict[int, int] = {}
    return [order.setdefault(int(c), len(order)) for c in raw]


class Embedder:
    """pyannote speaker embedding; returns a 1-D float32 vector."""

    def __init__(self, model_name: str, sample_rate: int = 16000):
        import torch
        from pyannote.audio import Inference, Model

        self.torch = torch
        self.sr = sample_rate
        model = Model.from_pretrained(model_name)
        self.inference = Inference(model, window="whole")

    def __call__(self, audio: np.ndarray) -> np.ndarray:
        wave = self.torch.from_numpy(np.ascontiguousarray(audio, dtype=np.float32))[None]
        return np.asarray(self.inference({"waveform": wave, "sample_rate": self.sr}), dtype=np.float32).reshape(-1)
```

import re

import numpy as np

PROMPT_MAX = 600  # chars; whisper keeps ~224 tokens of prompt


def clean_text(text: str, hallucinations: list[str]) -> str:
    out = text
    for h in hallucinations:
        out = re.sub(re.escape(h) + r"[.…!]*", "", out, flags=re.IGNORECASE)
    return re.sub(r"\s+", " ", out).strip()


def build_prompt(participants: list[str], glossary: list[str]) -> str:
    names = ["Владислав"] + [p for p in participants if p and p != "Владислав"]
    parts = [f"Участники: {', '.join(names)}."]
    if glossary:
        parts.append(f"Термины: {', '.join(glossary)}.")
    prompt = " ".join(parts)
    return prompt if len(prompt) <= PROMPT_MAX else prompt[:PROMPT_MAX].rsplit(",", 1)[0] + "."

### Task 9: Pipeline — step() and finalize()

**Files:**
- Create: `worker/transcriber/pipeline.py`
- Test: `worker/tests/test_pipeline.py`

Rules:
- Each `step()` reads everything available on both channels, segments, transcribes, labels, checks echo, writes Raw, saves state. Returns True if any record changed.
- Offsets saved = `segmenter.boundary()`; on load the readers start there. So the pipeline is idempotent across restarts.
- Mic utterances go to `pending_mic` until the sys channel is `echo_wait_s` past their end and no sys utterance is open that started before their end; then echo check → keep or drop (drop logged to `errors`? no: to logger only).
- Sys utterances ≥ `min_embed_s` get an embedding + online label; shorter ones inherit the last sys label (or "Speaker 1").
- ASR failure → `Rec.text = None`, message appended to `state.errors`.
- `finalize()`: drain remaining audio, force-flush pending mic, recluster all embeddings, relabel, write Raw with `финализирован HH:MM`, status `finalized`, delete `mic.pcm`/`sys.pcm` unless `keep_audio`.

from __future__ import annotations

import logging
from datetime import datetime
from typing import Callable

import numpy as np

from .asr import build_prompt
from .config import Config
from .echo import is_echo
from .pcm import PcmReader
from .session import Rec, Session
from .speakers import OnlineClusterer, label, recluster
from .vad import Segmenter, Utterance
from .vault import ensure_stub, write_raw

log = logging.getLogger("transcriber")

Asr = Callable[[np.ndarray, str], str]
Embed = Callable[[np.ndarray], np.ndarray]
ProbFactory = Callable[[], Callable[[np.ndarray], float]]


class Pipeline:
    def __init__(self, cfg: Config, session: Session, asr: Asr, embed: Embed, prob_factory: ProbFactory):
        self.cfg = cfg
        self.s = session
        self.asr = asr
        self.embed = embed
        st = session.state
        self.readers = {"mic": PcmReader(session.mic_pcm, st.mic_offset), "sys": PcmReader(session.sys_pcm, st.sys_offset)}
        self.segs = {
            ch: Segmenter(prob_factory(), cfg.sample_rate, cfg.vad_threshold, cfg.min_silence_ms, cfg.speech_pad_ms,
                          cfg.max_utterance_s, start_sample=off)
            for ch, off in (("mic", st.mic_offset), ("sys", st.sys_offset))
        }
        self.clusterer = OnlineClusterer.from_history(cfg.speaker_similarity, st.embeddings, st.speaker_of_emb)
        self.prompt = build_prompt(session.meta.participants, cfg.glossary)
        self.last_sys_speaker = next((r.speaker for r in reversed(st.records) if r.channel == "sys"), label(0))
        ensure_stub(cfg, session)
        if st.raw_path is None:
            write_raw(cfg, session, "идёт запись")
            session.save_state()

    # ---- live -------------------------------------------------------------

    def step(self) -> bool:
        changed = False
        for ch in ("mic", "sys"):
            audio = self.readers[ch].read_all()
            if audio.size:
                for u in self.segs[ch].feed(audio):
                    self._handle(ch, u)
                    changed = True
        changed |= self._flush_pending_mic(force=False)
        self.s.state.mic_offset = self.segs["mic"].boundary()
        self.s.state.sys_offset = self.segs["sys"].boundary()
        if changed:
            write_raw(self.cfg, self.s, "идёт запись")
        self.s.save_state()
        return changed

    def _sec(self, samples: int) -> float:
        return samples / self.cfg.sample_rate

    def _transcribe(self, u: Utterance) -> str | None:
        try:
            return self.asr(u.audio, self.prompt)
        except Exception as e:  # noqa: BLE001 — any model failure becomes a marker in the transcript
            msg = f"asr failed at {self._sec(u.start):.1f}-{self._sec(u.end):.1f}: {e!r}"
            log.warning(msg)
            self.s.state.errors.append(msg)
            return None

    def _handle(self, ch: str, u: Utterance) -> None:
        text = self._transcribe(u)
        rec = Rec(channel=ch, start=self._sec(u.start), end=self._sec(u.end), text=text, speaker="Me", emb=None)
        if ch == "mic":
            self.s.state.pending_mic.append(rec)
            return
        st = self.s.state
        if self._sec(u.samples) >= self.cfg.min_embed_s:
            emb = self.embed(u.audio)
            idx = self.clusterer.assign(emb)
            st.embeddings.append([float(x) for x in emb])
            st.speaker_of_emb.append(idx)
            rec.emb = len(st.embeddings) - 1
            rec.speaker = label(idx)
            self.last_sys_speaker = rec.speaker
        else:
            rec.speaker = self.last_sys_speaker
        st.records.append(rec)

    def _flush_pending_mic(self, force: bool) -> bool:
        st = self.s.state
        seg = self.segs["sys"]
        horizon = self._sec(seg.pos)
        open_start = self._sec(seg.open_start) if seg.open_start is not None else None
        sys_recs = [r for r in st.records if r.channel == "sys"]
        keep: list[Rec] = []
        changed = False
        for rec in st.pending_mic:
            ready = force or (horizon >= rec.end + self.cfg.echo_wait_s and (open_start is None or open_start > rec.end))
            if not ready:
                keep.append(rec)
                continue
            if is_echo(rec, sys_recs, self.cfg.echo_overlap, self.cfg.echo_similarity):
                log.info("echo dropped %.1f-%.1f: %s", rec.start, rec.end, rec.text)
            else:
                st.records.append(rec)
            changed = True
        st.pending_mic = keep
        return changed

    # ---- finalization -----------------------------------------------------

    def finalize(self) -> None:
        while self.step():
            pass
        self._flush_pending_mic(force=True)
        st = self.s.state
        if len(st.embeddings) >= 2:
            new = recluster(np.asarray(st.embeddings, dtype=np.float32), self.cfg.speaker_similarity)
            st.speaker_of_emb = new
            last = label(0)
            for r in sorted(st.records, key=lambda r: r.start):
                if r.channel != "sys":
                    continue
                if r.emb is not None:
                    r.speaker = label(new[r.emb])
                    last = r.speaker
                else:
                    r.speaker = last
        st.status = "finalized"
        write_raw(self.cfg, self.s, f"финализирован {datetime.now().strftime('%H:%M')}")
        self.s.save_state()
        if not self.cfg.keep_audio:
            for p in (self.s.mic_pcm, self.s.sys_pcm):
                p.unlink(missing_ok=True)

from __future__ import annotations

import logging
import time
from pathlib import Path

from .config import Config
from .models import Models, load_models
from .pipeline import Pipeline
from .session import Session

log = logging.getLogger("transcriber")


class Daemon:
    def __init__(self, cfg: Config, models: Models | None = None):
        self.cfg = cfg
        self._models = models
        self.pipes: dict[Path, Pipeline] = {}

    @property
    def models(self) -> Models:
        if self._models is None:
            log.info("loading models")
            self._models = load_models(self.cfg)
        return self._models

    def once(self) -> bool:
        """One pass over all sessions. Returns True if any session did work."""
        worked = False
        if not self.cfg.sessions_dir.exists():
            return False
        for d in sorted(p for p in self.cfg.sessions_dir.iterdir() if (p / "meta.json").exists()):
            try:
                pipe = self.pipes.get(d)
                if pipe is None:
                    session = Session.load(d)
                    if session.state.status == "finalized":
                        continue
                    m = self.models
                    pipe = Pipeline(self.cfg, session, m.asr, m.embed, m.prob_factory)
                    self.pipes[d] = pipe
                worked |= pipe.step()
                if pipe.s.stop_requested():
                    log.info("finalizing %s", d.name)
                    pipe.finalize()
                    del self.pipes[d]
                    worked = True
            except Exception:  # noqa: BLE001 — one broken session must not stop the others
                log.exception("session %s failed", d.name)
                self.pipes.pop(d, None)
        return worked

    def run_forever(self) -> None:
        log.info("worker started, sessions at %s", self.cfg.sessions_dir)
        while True:
            worked = self.once()
            time.sleep(0.2 if worked else 1.0)
```

from __future__ import annotations

import argparse
import json
import logging
import math
import sys
import wave
from datetime import datetime
from pathlib import Path

import numpy as np

from .config import Config, load_config
from .daemon import Daemon
from .models import Models, load_models

from .session import Session


def read_wav(path: Path, target_sr: int) -> np.ndarray:
    with wave.open(str(path), "rb") as w:
        sr, ch, width, n = w.getframerate(), w.getnchannels(), w.getsampwidth(), w.getnframes()
        raw = w.readframes(n)
    if width != 2:
        raise SystemExit(f"{path}: only 16-bit WAV supported")
    audio = np.frombuffer(raw, dtype=np.int16).astype(np.float32) / 32768.0
    if ch > 1:
        audio = audio.reshape(-1, ch).mean(axis=1)
    if sr != target_sr:
        from scipy.signal import resample_poly

        g = math.gcd(sr, target_sr)
        audio = resample_poly(audio, target_sr // g, sr // g).astype(np.float32)
    return audio


def from_wav(cfg: Config, mic: Path, sys_: Path, title: str, app: str, models: Models | None = None,
             started_at: datetime | None = None) -> Path:
    started_at = started_at or datetime.now().astimezone()
    sid = started_at.strftime("%Y-%m-%d-%H%M%S")
    d = cfg.sessions_dir / sid
    d.mkdir(parents=True, exist_ok=True)
    (d / "meta.json").write_text(json.dumps({"id": sid, "started_at": started_at.isoformat(), "app": app,
                                             "title": title, "participants": []}, ensure_ascii=False))
    for name, src in (("mic.pcm", mic), ("sys.pcm", sys_)):
        (d / name).write_bytes((np.clip(read_wav(src, cfg.sample_rate), -1, 1) * 32767).astype(np.int16).tobytes())
    (d / "STOP").touch()
    m = models or load_models(cfg)
    pipe = Pipeline(cfg, Session.load(d), m.asr, m.embed, m.prob_factory)
    pipe.finalize()
    return Path(pipe.s.state.raw_path)


def main(argv: list[str] | None = None) -> None:
    logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
    ap = argparse.ArgumentParser(prog="transcriber")
    sub = ap.add_subparsers(dest="cmd", required=True)
    sub.add_parser("run", help="daemon: watch sessions dir")
    fw = sub.add_parser("from-wav", help="transcribe two WAV files as one session")
    fw.add_argument("mic")
    fw.add_argument("sys")
    fw.add_argument("--title", default="")
    fw.add_argument("--app", default="manual")
    fz = sub.add_parser("finalize", help="finalize a session by id")
    fz.add_argument("session_id")
    args = ap.parse_args(argv)
    cfg = load_config()
    if args.cmd == "run":
        Daemon(cfg).run_forever()
    elif args.cmd == "from-wav":
        print(from_wav(cfg, Path(args.mic), Path(args.sys), args.title, args.app))
    elif args.cmd == "finalize":
        d = cfg.sessions_dir / args.session_id
        (d / "STOP").touch()
        m = load_models(cfg)
        pipe = Pipeline(cfg, Session.load(d), m.asr, m.embed, m.prob_factory)
        pipe.finalize()
        print(pipe.s.state.raw_path)


if __name__ == "__main__":
    main(sys.argv[1:])
