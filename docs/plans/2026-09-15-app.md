# Transcriber Mac App Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A menubar Mac app that notices a meeting in Толк/Zoom, offers to record, captures mic + system audio into a session dir the Python worker already consumes, keeps the Mac awake while recording, shows a floating button that opens the meeting note in Obsidian, and stops when the meeting ends.

**Architecture:** SwiftPM package with two targets. `TranscriberCore` (library, unit-tested with swift-testing) holds everything without a window: meeting detection state machine, Core Audio process listing, process-tap capture, float→int16/16 kHz downmix, append-only PCM writer, session directory protocol (`meta.json`, `STOP`), power holds, Obsidian URL, Толк inspection through Accessibility. `Transcriber` (executable, AppKit, manually tested) wires the core into a status item, a prompt panel, and a floating button. The app is bundled by a script into `Transcriber.app`, signed with the self-signed identity «Transcriber Dev» (already in the login keychain; TCC grants for mic and system audio are already given to bundle id `ru.slesarev.transcriber` signed with it), installed to `~/Applications`, and started at login by launchd.

**Tech Stack:** Swift 6.1 toolchain from Command Line Tools (no Xcode), SwiftPM tools-version 5.9 (Swift 5 language mode), swift-testing, AppKit, CoreAudio process taps (macOS 14.2+), AVFoundation `AVAudioConverter`, IOKit power assertions, ApplicationServices (AX).

Spec: `docs/specs/2026-09-15-transcriber-design.md`. Worker contract (`worker/transcriber/session.py`, `vault.py`): the app creates `sessions/<id>/meta.json` with `id`, `started_at` (ISO 8601 with local offset, e.g. `2026-09-15T14:30:00+03:00`), `app`, `title`, `participants`; writes `mic.pcm` and `sys.pcm` (int16 mono 16 kHz, headerless); creates an empty `STOP` file to end the session. The worker writes `state.json` with `note_path` and `raw_path`; the app reads `note_path` to open the note.

Verified on this machine during planning (see `docs/plans/probes/` after Task 0): process list + `kAudioProcessPropertyIsRunningInput` works; `CATapDescription(stereoGlobalTapButExcludeProcesses:)` + aggregate device with the default mic yields two input buffers `[mic 1ch, tap 2ch interleaved]` at 48 kHz float32; a signed bundle got both TCC grants; `swift test` works with `import Testing` (not XCTest).

```swift
import Foundation

public final class Log {
    public static let shared = Log(file: FileManager.default.homeDirectoryForCurrentUser
        .appendingPathComponent("Library/Logs/transcriber-app.log"))

    private let file: URL
    private let queue = DispatchQueue(label: "transcriber.log")
    private let formatter: DateFormatter = {
        let f = DateFormatter()
        f.dateFormat = "yyyy-MM-dd HH:mm:ss"
        return f
    }()

    public init(file: URL) {
        self.file = file
    }

    public func info(_ message: String) { write("INFO", message) }
    public func warn(_ message: String) { write("WARN", message) }
    public func error(_ message: String) { write("ERROR", message) }

    private func write(_ level: String, _ message: String) {
        let line = "\(formatter.string(from: Date())) \(level) \(message)\n"
        queue.sync {
            if let handle = try? FileHandle(forWritingTo: file) {
                handle.seekToEndOfFile()
                handle.write(line.data(using: .utf8)!)
                try? handle.close()
            } else {
                try? line.write(to: file, atomically: true, encoding: .utf8)
            }
        }
    }
}
```

```swift
import Foundation

public struct SessionDir: Equatable {
    public let url: URL
    public var id: String { url.lastPathComponent }
    public var metaURL: URL { url.appendingPathComponent("meta.json") }
    public var stateURL: URL { url.appendingPathComponent("state.json") }
    public var stopURL: URL { url.appendingPathComponent("STOP") }
    public var micURL: URL { url.appendingPathComponent("mic.pcm") }
    public var sysURL: URL { url.appendingPathComponent("sys.pcm") }

    public func requestStop() throws {
        if !FileManager.default.fileExists(atPath: stopURL.path) {
            try Data().write(to: stopURL)
        }
    }

    private func state() -> [String: Any]? {
        guard let data = try? Data(contentsOf: stateURL) else { return nil }
        return (try? JSONSerialization.jsonObject(with: data)) as? [String: Any]
    }

    public func notePath() -> String? { state()?["note_path"] as? String }
    public func status() -> String? { state()?["status"] as? String }
}

public final class SessionStore {
    public let root: URL

    public init(root: URL) {
        self.root = root
    }

    public static var defaultRoot: URL {
        FileManager.default.homeDirectoryForCurrentUser.appendingPathComponent("Claude/transcriber/sessions")
    }

    public func create(app: String, title: String, participants: [String], now: Date = Date()) throws -> SessionDir {
        try FileManager.default.createDirectory(at: root, withIntermediateDirectories: true)
        let base = Self.idFormatter.string(from: now)
        var id = base
        var n = 0
        while FileManager.default.fileExists(atPath: root.appendingPathComponent(id).path) {
            n += 1
            id = "\(base)-\(n)"
        }
        let dir = SessionDir(url: root.appendingPathComponent(id))
        try FileManager.default.createDirectory(at: dir.url, withIntermediateDirectories: true)
        let meta: [String: Any] = [
            "id": id,
            "started_at": Self.isoFormatter.string(from: now),
            "app": app,
            "title": title,
            "participants": participants,
        ]
        let data = try JSONSerialization.data(withJSONObject: meta, options: [.prettyPrinted, .withoutEscapingSlashes])
        try data.write(to: dir.metaURL, options: .atomic)
        return dir
    }

    public func open(id: String) -> SessionDir? {
        let dir = SessionDir(url: root.appendingPathComponent(id))
        return FileManager.default.fileExists(atPath: dir.metaURL.path) ? dir : nil
    }

    private static let idFormatter: DateFormatter = {
        let f = DateFormatter()
        f.locale = Locale(identifier: "en_US_POSIX")
        f.dateFormat = "yyyy-MM-dd-HHmm"
        return f
    }()

    /// ISO 8601 with the local offset (`+03:00`), which the worker uses for vault filenames.
    private static let isoFormatter: DateFormatter = {
        let f = DateFormatter()
        f.locale = Locale(identifier: "en_US_POSIX")
        f.dateFormat = "yyyy-MM-dd'T'HH:mm:ssXXX"
        return f
    }()
}
```

```swift
import Foundation

/// Append-only int16 mono PCM file. `write` buffers; every `flushEvery` samples (default one second
/// at 16 kHz) the buffer is written and fsync'ed, so a lid close loses at most one second.
public final class PcmWriter {
    private let handle: FileHandle
    private let flushEvery: Int
    private var buffer: [Int16] = []
    public private(set) var samplesWritten = 0

    public init(url: URL, flushEvery: Int = 16_000) throws {
        if !FileManager.default.fileExists(atPath: url.path) {
            FileManager.default.createFile(atPath: url.path, contents: nil)
        }
        handle = try FileHandle(forWritingTo: url)
        try handle.seekToEnd()
        self.flushEvery = flushEvery
    }

    public func write(_ samples: [Int16]) {
        buffer.append(contentsOf: samples)
        if buffer.count >= flushEvery {
            flush()
        }
    }

    public func flush() {
        guard !buffer.isEmpty else { return }
        let data = buffer.withUnsafeBufferPointer { Data(buffer: $0) }
        handle.write(data)
        fsync(handle.fileDescriptor)
        samplesWritten += buffer.count
        buffer.removeAll(keepingCapacity: true)
    }

    public func close() {
        flush()
        try? handle.close()
    }
}
```

```swift
    let n = Int(rate * seconds)
    var out = [Float](repeating: 0, count: n * channels)

    }
    return out
}

private func zeroCrossings(_ x: [Int16]) -> Int {
```

```swift
import AVFoundation
import Foundation

public enum DownmixerError: Error { case badFormat, converterUnavailable }

/// Converts interleaved float32 audio at any rate/channel count to int16 mono 16 kHz.
/// Not thread-safe: call from one serial queue.
public final class Downmixer {
    public static let outputRate: Double = 16_000

    private let inFormat: AVAudioFormat
    private let outFormat: AVAudioFormat
    private let converter: AVAudioConverter
    private let ratio: Double

    public init(inputRate: Double, inputChannels: Int) throws {
        guard let inF = AVAudioFormat(commonFormat: .pcmFormatFloat32, sampleRate: inputRate,
                                      channels: AVAudioChannelCount(inputChannels), interleaved: true),
              let outF = AVAudioFormat(commonFormat: .pcmFormatInt16, sampleRate: Self.outputRate,
                                       channels: 1, interleaved: true)
        else { throw DownmixerError.badFormat }
        guard let conv = AVAudioConverter(from: inF, to: outF) else { throw DownmixerError.converterUnavailable }
        conv.downmix = true
        inFormat = inF
        outFormat = outF
        converter = conv
        ratio = Self.outputRate / inputRate
    }

    /// `interleaved` holds `frames * inputChannels` floats.
    public func process(interleaved: [Float], frames: Int) -> [Int16] {
        guard frames > 0,
              let inBuf = AVAudioPCMBuffer(pcmFormat: inFormat, frameCapacity: AVAudioFrameCount(frames))
        else { return [] }
        inBuf.frameLength = AVAudioFrameCount(frames)
        interleaved.withUnsafeBufferPointer { src in
            let dst = inBuf.audioBufferList.pointee.mBuffers.mData!.assumingMemoryBound(to: Float.self)
            dst.update(from: src.baseAddress!, count: frames * Int(inFormat.channelCount))
        }
        return convert(input: inBuf, capacity: Int(Double(frames) * ratio) + 64)
    }

    /// Drains the converter's internal delay line. Call once at the end of a recording.
    public func finish() -> [Int16] {
        let out = convert(input: nil, capacity: 4096)
        converter.reset()
        return out
    }

    private func convert(input: AVAudioPCMBuffer?, capacity: Int) -> [Int16] {
        guard let outBuf = AVAudioPCMBuffer(pcmFormat: outFormat, frameCapacity: AVAudioFrameCount(capacity)) else { return [] }
        var delivered = false
        var error: NSError?
        let status = converter.convert(to: outBuf, error: &error) { _, outStatus in
            if delivered || input == nil {
                outStatus.pointee = input == nil ? .endOfStream : .noDataNow
                return nil
            }
            delivered = true
            outStatus.pointee = .haveData
            return input
        }
        if status == .error {
            Log.shared.error("downmix: \(error?.localizedDescription ?? "unknown")")
            return []
        }
        let n = Int(outBuf.frameLength)
        let p = outBuf.audioBufferList.pointee.mBuffers.mData!.assumingMemoryBound(to: Int16.self)
        return Array(UnsafeBufferPointer(start: p, count: n))
    }
}
```

```swift
import Foundation

public struct AudioProcess: Equatable {
    public let objectID: AudioObjectID
    public let pid: pid_t
    public let bundleID: String
    public let isRunningInput: Bool

    public init(objectID: AudioObjectID, pid: pid_t, bundleID: String, isRunningInput: Bool) {
        self.objectID = objectID
        self.pid = pid
        self.bundleID = bundleID
        self.isRunningInput = isRunningInput
    }
}

enum CoreAudioProps {
    static func address(_ selector: AudioObjectPropertySelector,
                        scope: AudioObjectPropertyScope = kAudioObjectPropertyScopeGlobal) -> AudioObjectPropertyAddress {
        AudioObjectPropertyAddress(mSelector: selector, mScope: scope, mElement: kAudioObjectPropertyElementMain)
    }

    static func get<T>(_ object: AudioObjectID, _ selector: AudioObjectPropertySelector,
                       scope: AudioObjectPropertyScope = kAudioObjectPropertyScopeGlobal, default value: T) -> T {
        var addr = address(selector, scope: scope)
        var size = UInt32(MemoryLayout<T>.size)
        var out = value
        let status = AudioObjectGetPropertyData(object, &addr, 0, nil, &size, &out)
        return status == noErr ? out : value
    }

    static func getString(_ object: AudioObjectID, _ selector: AudioObjectPropertySelector) -> String {
        var addr = address(selector)
        var size = UInt32(MemoryLayout<CFString?>.size)
        var value: Unmanaged<CFString>? = nil
        let status = AudioObjectGetPropertyData(object, &addr, 0, nil, &size, &value)
        guard status == noErr, let v = value?.takeRetainedValue() else { return "" }
        return v as String
    }

    static func getArray<T>(_ object: AudioObjectID, _ selector: AudioObjectPropertySelector, of: T.Type) -> [T] {
        var addr = address(selector)
        var size: UInt32 = 0
        guard AudioObjectGetPropertyDataSize(object, &addr, 0, nil, &size) == noErr, size > 0 else { return [] }
        var items = [T](repeating: unsafeBitCast(0 as UInt32, to: T.self), count: Int(size) / MemoryLayout<T>.size)
        guard AudioObjectGetPropertyData(object, &addr, 0, nil, &size, &items) == noErr else { return [] }
        return items
    }
}

public enum AudioProcesses {
    /// Every process Core Audio knows about, with whether it currently records from an input device.
    public static func list() -> [AudioProcess] {
```

```swift
public enum MeetingEvent: Equatable {
    case started(app: String)
    case ended
}

/// Polled state machine: a meeting starts when a watched app records from the mic, and ends
/// when no watched app has used the mic for `endGrace` seconds.
public final class MeetingDetector {
    public static let defaultApps = ["kontur.talk", "us.zoom"]

    private let apps: [String]
    private let lister: () -> [AudioProcess]
    private let endGrace: TimeInterval
    public private(set) var activeApp: String?
    private var micLostAt: Date?

    public init(apps: [String] = MeetingDetector.defaultApps, lister: @escaping () -> [AudioProcess],
                endGrace: TimeInterval = 60) {
        self.apps = apps
        self.lister = lister
        self.endGrace = endGrace
    }

    /// Returns the watched app (by prefix) that currently records from the mic, if any.
    private func appUsingMic() -> String? {
        for p in lister() where p.isRunningInput {
            if let app = apps.first(where: { p.bundleID.hasPrefix($0) }) { return app }
        }
        return nil
    }

    public func poll(now: Date) -> MeetingEvent? {
        let using = appUsingMic()
        if activeApp == nil {
            guard let app = using else { return nil }
            activeApp = app
            micLostAt = nil
            return .started(app: app)
        }
        if using != nil {
            micLostAt = nil
            return nil
        }
        if let lost = micLostAt {
            if now.timeIntervalSince(lost) >= endGrace {
                activeApp = nil
                micLostAt = nil
                return .ended
            }
            return nil
        }
        micLostAt = now
        return nil
    }

    public func reset() {
        activeApp = nil
        micLostAt = nil
    }
}
```

```swift
import AudioToolbox
import CoreAudio
import Foundation

public enum CaptureError: Error, CustomStringConvertible {
    case coreAudio(String, OSStatus)
    case noInputDevice
    case unexpectedStreams(Int)

    public var description: String {
        switch self {
        case .coreAudio(let what, let status): return "\(what) failed: OSStatus \(status)"
        case .noInputDevice: return "no default input device"
        case .unexpectedStreams(let n): return "aggregate device has \(n) input buffers, expected 2 (mic, tap)"
        }
    }
}

/// One IOProc cycle's worth of audio from both inputs, deinterleaved by role.
public struct CaptureChunk {
    public let mic: [Float]     // mono
    public let sys: [Float]     // interleaved, `sysChannels` channels
    public let frames: Int
}

/// Captures the default microphone and a global process tap (all processes except ours)
/// through a private aggregate device. Delivers chunks on a serial queue.
public final class TapCapture {
    public private(set) var sampleRate: Double = 48_000
    public private(set) var sysChannels: Int = 2

    private var tapID = AudioObjectID(kAudioObjectUnknown)
    private var aggregateID = AudioObjectID(kAudioObjectUnknown)
    private var procID: AudioDeviceIOProcID?
    private let queue = DispatchQueue(label: "transcriber.capture", qos: .userInitiated)
    private var handler: ((CaptureChunk) -> Void)?

    public init() {}

    public func start(onChunk: @escaping (CaptureChunk) -> Void) throws {
        handler = onChunk
        let system = AudioObjectID(kAudioObjectSystemObject)
        let mic: AudioObjectID = CoreAudioProps.get(system, kAudioHardwarePropertyDefaultInputDevice, default: AudioObjectID(0))
        guard mic != 0 else { throw CaptureError.noInputDevice }
        let micUID = CoreAudioProps.getString(mic, kAudioDevicePropertyDeviceUID)

        let excluded = AudioProcesses.ownProcessObject().map { [$0] } ?? []
        let desc = CATapDescription(stereoGlobalTapButExcludeProcesses: excluded)
        desc.name = "transcriber"
        desc.isPrivate = true
        desc.muteBehavior = .unmuted
        var tap = AudioObjectID(kAudioObjectUnknown)
        var status = AudioHardwareCreateProcessTap(desc, &tap)
        guard status == noErr else { throw CaptureError.coreAudio("AudioHardwareCreateProcessTap", status) }
        tapID = tap

        let aggDesc: [String: Any] = [
            kAudioAggregateDeviceNameKey as String: "transcriber-agg",
            kAudioAggregateDeviceUIDKey as String: "ru.slesarev.transcriber.agg",
            kAudioAggregateDeviceIsPrivateKey as String: true,
            kAudioAggregateDeviceIsStackedKey as String: false,
            kAudioAggregateDeviceTapAutoStartKey as String: true,
            kAudioAggregateDeviceSubDeviceListKey as String: [[kAudioSubDeviceUIDKey as String: micUID]],
            kAudioAggregateDeviceTapListKey as String: [[
                kAudioSubTapUIDKey as String: desc.uuid.uuidString,
                kAudioSubTapDriftCompensationKey as String: true,
            ]],
        ]
        var agg = AudioObjectID(kAudioObjectUnknown)
        status = AudioHardwareCreateAggregateDevice(aggDesc as CFDictionary, &agg)
        guard status == noErr else {
            AudioHardwareDestroyProcessTap(tapID)
            throw CaptureError.coreAudio("AudioHardwareCreateAggregateDevice", status)
        }
        aggregateID = agg
        sampleRate = CoreAudioProps.get(agg, kAudioDevicePropertyNominalSampleRate, default: 48_000.0)

        // Input layout: sub-devices first (mic, 1 buffer), then taps (1 buffer, stereo interleaved).
        let channels = inputBufferChannels(agg)
        guard channels.count == 2 else {
            teardown()
            throw CaptureError.unexpectedStreams(channels.count)
        }
        sysChannels = channels[1]
        Log.shared.info("capture: rate \(sampleRate), mic \(channels[0])ch, tap \(channels[1])ch")

        var proc: AudioDeviceIOProcID?
        status = AudioDeviceCreateIOProcIDWithBlock(&proc, agg, queue) { [weak self] _, inData, _, _, _ in
            self?.deliver(inData)
        }
        guard status == noErr, let p = proc else {
            teardown()
            throw CaptureError.coreAudio("AudioDeviceCreateIOProcIDWithBlock", status)
        }
        procID = p
        status = AudioDeviceStart(agg, p)
        guard status == noErr else {
            teardown()
            throw CaptureError.coreAudio("AudioDeviceStart", status)
        }
    }

    public func stop() {
        if let p = procID, aggregateID != kAudioObjectUnknown {
            AudioDeviceStop(aggregateID, p)
        }
        teardown()
        queue.sync {}   // let the last delivered chunk finish
        handler = nil
    }

    private func teardown() {
        if let p = procID, aggregateID != kAudioObjectUnknown {
            AudioDeviceDestroyIOProcID(aggregateID, p)
        }
        procID = nil
        if aggregateID != kAudioObjectUnknown {
            AudioHardwareDestroyAggregateDevice(aggregateID)
            aggregateID = kAudioObjectUnknown
        }
        if tapID != kAudioObjectUnknown {
            AudioHardwareDestroyProcessTap(tapID)
            tapID = kAudioObjectUnknown
        }
    }

    private func inputBufferChannels(_ device: AudioObjectID) -> [Int] {
        var addr = CoreAudioProps.address(kAudioDevicePropertyStreamConfiguration, scope: kAudioObjectPropertyScopeInput)
        var size: UInt32 = 0
        guard AudioObjectGetPropertyDataSize(device, &addr, 0, nil, &size) == noErr else { return [] }
        let list = UnsafeMutablePointer<AudioBufferList>.allocate(capacity: Int(size))
        defer { list.deallocate() }
        guard AudioObjectGetPropertyData(device, &addr, 0, nil, &size, list) == noErr else { return [] }
        return UnsafeMutableAudioBufferListPointer(list).map { Int($0.mNumberChannels) }
    }

    /// Runs on `queue` (the IOProc dispatch queue): copy out and hand over.
    private func deliver(_ inData: UnsafePointer<AudioBufferList>) {
        let buffers = UnsafeMutableAudioBufferListPointer(UnsafeMutablePointer(mutating: inData))
        guard buffers.count >= 2, let handler else { return }
        let micBuf = buffers[0]
        let sysBuf = buffers[1]
        let micCount = Int(micBuf.mDataByteSize) / MemoryLayout<Float>.size
        let sysCount = Int(sysBuf.mDataByteSize) / MemoryLayout<Float>.size
        let mic = Array(UnsafeBufferPointer(start: micBuf.mData?.assumingMemoryBound(to: Float.self), count: micCount))
        let sys = Array(UnsafeBufferPointer(start: sysBuf.mData?.assumingMemoryBound(to: Float.self), count: sysCount))
        let frames = micCount / max(Int(micBuf.mNumberChannels), 1)
        handler(CaptureChunk(mic: mic, sys: sys, frames: frames))
    }
}
```

```swift
import IOKit.pwr_mgt

/// Keeps the Mac awake while recording: a power assertion against idle sleep, plus (when the
/// sudoers rule exists) `pmset disablesleep` so a closed lid does not sleep either.
public final class PowerHold {
    private var assertion: IOPMAssertionID = 0
    private var held = false
    private let usePmset: Bool

    public init(usePmset: Bool = true) {
        self.usePmset = usePmset
    }

    public func acquire() {
        guard !held else { return }
        let status = IOPMAssertionCreateWithName(kIOPMAssertionTypePreventUserIdleSystemSleep as CFString,
                                                 IOPMAssertionLevel(kIOPMAssertionLevelOn),
                                                 "Транскрибатор пишет встречу" as CFString, &assertion)
        if status != kIOReturnSuccess { Log.shared.warn("power assertion failed: \(status)") }
        held = true
        if usePmset { pmset(disableSleep: true) }
    }

    public func release() {
        guard held else { return }
        IOPMAssertionRelease(assertion)
        assertion = 0
        held = false
        if usePmset { pmset(disableSleep: false) }
    }

    private func pmset(disableSleep: Bool) {
        let p = Process()
        p.executableURL = URL(fileURLWithPath: "/usr/bin/sudo")
        p.arguments = ["-n", "/usr/bin/pmset", "-a", "disablesleep", disableSleep ? "1" : "0"]
        p.standardOutput = FileHandle.nullDevice
        p.standardError = FileHandle.nullDevice
        do {
            try p.run()
            p.waitUntilExit()
            if p.terminationStatus != 0 {
                Log.shared.warn("pmset disablesleep \(disableSleep ? 1 : 0) skipped (exit \(p.terminationStatus)); add the sudoers rule to enable it")
            } else {
                Log.shared.info("pmset disablesleep \(disableSleep ? 1 : 0)")
            }
        } catch {
            Log.shared.warn("pmset launch failed: \(error)")
        }
    }
}
```

```swift
    public static func participantNames(from texts: [String]) -> [String] {
        var seen = Set<String>()
        return texts.compactMap { raw in
            let t = raw.trimmingCharacters(in: .whitespacesAndNewlines)
            let range = NSRange(t.startIndex..., in: t)
            guard namePattern.firstMatch(in: t, range: range) != nil, !seen.contains(t) else { return nil }
            seen.insert(t)
            return t
        }

    private static func stringAttribute(_ element: AXUIElement, _ name: String) -> String {
        var value: CFTypeRef?
```

```swift
    private let power = PowerHold()
    private var capture: TapCapture?
    private var micMixer: Downmixer?
    private var sysMixer: Downmixer?
    private var micWriter: PcmWriter?
    private var sysWriter: PcmWriter?
    private let log = Log.shared

    init(store: SessionStore = SessionStore(root: SessionStore.defaultRoot)) {
        self.store = store
    }

    var current: SessionDir? {
        if case .recording(let dir) = state { return dir }
        return nil
    }

    func start(app: String) {
        guard case .idle = state else { return }
        let info = app == TalkInspector.bundleID ? TalkInspector.inspect() : MeetingInfo()
        do {
            let dir = try store.create(app: app, title: info.title, participants: info.participants)
            micWriter = try PcmWriter(url: dir.micURL)
            sysWriter = try PcmWriter(url: dir.sysURL)
            let cap = TapCapture()
            try cap.start { [weak self] chunk in self?.consume(chunk) }
            capture = cap
            sysMixer = try Downmixer(inputRate: cap.sampleRate, inputChannels: cap.sysChannels)
            micMixer = nil   // built on the first chunk, once the mic channel count is known
            power.acquire()
            state = .recording(dir)
            log.info("session \(dir.id) started (app \(app), title '\(info.title)')")
            onStateChange?(state)
        } catch {
            log.error("session start failed: \(error)")
            cleanup()
        }
    }

    func stop() {
        guard case .recording(let dir) = state else { return }
        capture?.stop()
        if let m = micMixer { micWriter?.write(m.finish()) }
        if let s = sysMixer { sysWriter?.write(s.finish()) }
        cleanup()
        do { try dir.requestStop() } catch { log.error("STOP write failed: \(error)") }
        power.release()
        state = .idle
        log.info("session \(dir.id) stopped; mic \(micWriter?.samplesWritten ?? 0) samples")
        onStateChange?(state)
    }

    private func cleanup() {
        capture = nil
        micWriter?.close()
        sysWriter?.close()
        micWriter = nil
        sysWriter = nil
        micMixer = nil
        sysMixer = nil
    }

    /// Runs on the capture queue.
    private func consume(_ chunk: CaptureChunk) {
        guard chunk.frames > 0, let capture else { return }
        if micMixer == nil {
            let channels = max(chunk.mic.count / chunk.frames, 1)
            micMixer = try? Downmixer(inputRate: capture.sampleRate, inputChannels: channels)
        }
        if let m = micMixer { micWriter?.write(m.process(interleaved: chunk.mic, frames: chunk.frames)) }
        if let s = sysMixer { sysWriter?.write(s.process(interleaved: chunk.sys, frames: chunk.frames)) }
    }
}
```

`stop()` races with `consume` on the capture queue: `TapCapture.stop()` stops the device and does `queue.sync {}` before returning, so no chunk is in flight when the writers close.

```swift
            bg.fill()
```

```swift
        NSWorkspace.shared.notificationCenter.addObserver(forName: NSWorkspace.didWakeNotification, object: nil,
                                                          queue: .main) { [weak self] _ in
            self?.log.info("woke from sleep")
            self?.poll()

        poll()

        controller.stop()

    private func poll() {
        guard let event = detector.poll(now: Date()) else { return }
        switch event {
        case .started(let app):
            log.info("meeting detected in \(app)")
            if controller.current != nil || declinedApp == app { return }
            let name = app == TalkInspector.bundleID ? "Толке" : "Zoom"
            prompt.show(appName: name,
                        onRecord: { [weak self] in self?.controller.start(app: app) },
                        onSkip: { [weak self] in self?.declinedApp = app })
        case .ended:
            log.info("meeting ended")
            declinedApp = nil
            prompt.dismiss()
            controller.stop()
        }
    }

    private func toggle() {
        if controller.current != nil {
            controller.stop()
            detector.reset()
        } else {
            prompt.dismiss()
            controller.start(app: detector.activeApp ?? "manual")
        }
```
